# Music Visualizer parameters
# Edit values after '=' and restart the application to apply them.

# Window and display
useFullScreen=false
canvasWidth=1920
canvasHeight=1080
# Zero-based display index preselected in the fullscreen display chooser.
visualizationDisplay=0
# The separate, resizable control window starts at these dimensions.
controlsWindowWidth=900
controlsWindowHeight=1080
fps=60

# Audio analysis
# Default driver selected at startup (minim or wasapi); choose any available
# driver and device in the startup dialog. File playback always uses Minim.
liveInputBackend=wasapi
# Default selection in the WASAPI startup dialog: shared or exclusive.
# Exclusive mode requires a compatible device and can prevent other applications from using it.
wasapiMode=shared
# Default buffer sizes in the live-input startup dialog. PortAudio capture read
# size is separate from the audio / FFT window; file playback uses audioBufferSize.
# Startup choices are powers of two: 32-8192 capture frames, 32-65536 audio samples.
captureBufferSize=256
audioBufferSize=512
# Color masks and low-to-high FFT-band mapping. Use unique initials: B=blue, C=cyan,
# R=red, O=orange, Y=yellow, G=green, M=magenta. The current setting uses BCROY.
colorBandMapping=BCGROY
maxFftFrequency=16000.0
frequencyMappingExponent=3.0
# Fraction (0 to 0.95) by which each band widens into its neighbours; 0 means no overlap.
bandOverlap=0.0
audioBandMultiplier=0.1
audioBloomMultiplier=1.0
# Solid and bloom mask intensity are composited with at most 10 image draws per mask.
# Increasing either multiplier beyond this effective intensity cap will not make masks brighter.
initialAudioSensitivity=2.0
audioBaseIntensityScale=0.25
baseImageSilenceFloor=0.0
audioAttackTimeMs=25.0
audioReleaseTimeMs=250.0

# Spotlight reveal mode (toggle at runtime with M)
# Scales audio level before it is converted into spotlight area.
spotlightEnergyMultiplier=1.0
# Quiet and loud radii, measured as fractions of image height.
spotlightMinimumRadius=0.04
spotlightMaximumRadius=0.5
# Fraction of each spotlight radius used for fading: 0 is sharp, 1 fades from center to edge.
spotlightEdgeFeatherProportion=0.35
# Milliseconds between new Brownian motion targets.
spotlightMotionUpdateIntervalMs=1000.0
# Scale of each random target displacement relative to image size.
spotlightBrownianStep=0.03
# Millisecond time constant for gliding toward a new target; 0 moves immediately.
spotlightMotionSmoothingTimeMs=1000.0
# Use a separate moving spotlight for every color mask.
spotlightIndependentMaskPositions=true
# Keep each spotlight circular by preventing clipping at image edges.
spotlightKeepCirclesInsideImage=false

# Area Reveal Mode
# Start up rendering mode (FULL, SPOTLIGHT, or AREA)
startRevealMode=FULL
# Scales audio level before it regulates pane light-up probability
areaEnergyMultiplier=1.0
# Seconds it takes for the random area selection to do a full migration cycle
areaCycleDurationSeconds=12.0
# Active-tile rendering: WHOLE_TILE, CENTER_GLOW, or TRAIL.
areaRevealStyle=WHOLE_TILE
